cmake_minimum_required(VERSION 3.25)
project(seashell_bakeoff_win32 CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(bakeoff-win32 WIN32 main.cpp)
target_link_libraries(bakeoff-win32 PRIVATE d2d1 dwrite)
target_compile_definitions(bakeoff-win32 PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(bakeoff-win32 PRIVATE /W3 /utf-8)
endif()
